## 💻 4. Code Source Complet

### A. Le Bloc React (`src/editor/blocks/LawBlock.tsx`)

```tsx
import { createReactBlockSpec } from "@blocknote/react";

export const LawBlock = createReactBlockSpec(
  {
    type: "law",
    propSchema: {
      codeName: { default: "Code de la commande publique" },
      articleNumber: { default: "Article L. 2122-1" },
      legifranceUrl: {
        default:
          "https://www.legifrance.gouv.fr/codes/article_lc/LEGIARTI000037728472",
      },
      excerpt: {
        default:
          "L'acheteur peut passer un marché sans publicité ni mise en concurrence préalables...",
      },
      status: { default: "En vigueur" },
    },
    content: "none",
  },
  {
    render: (props) => {
      const { codeName, articleNumber, legifranceUrl, excerpt, status } =
        props.block.props;

      return (
        <div className="fr-callout fr-callout--blue-france my-4 p-4 border-l-4 border-[#000091] bg-blue-50/50 dark:bg-blue-950/20 rounded-r-lg shadow-xs">
          <div className="flex items-center justify-between gap-2 mb-2">
            <div className="flex items-center gap-2">
              <span className="text-xl">⚖️</span>
              <span className="text-sm font-bold text-[#000091] dark:text-[#8585f6]">
                {codeName} — {articleNumber}
              </span>
            </div>
            <span className="text-xs px-2 py-0.5 rounded bg-emerald-100 text-emerald-800 dark:bg-emerald-950 dark:text-emerald-300 font-semibold">
              {status}
            </span>
          </div>

          <blockquote className="text-sm italic text-gray-700 dark:text-gray-300 my-2 border-l-2 border-gray-300 dark:border-gray-700 pl-3">
            « {excerpt} »
          </blockquote>

          <div className="flex items-center justify-end pt-2">
            <a
              href={legifranceUrl}
              target="_blank"
              rel="noopener noreferrer"
              className="text-xs font-semibold text-[#000091] dark:text-[#8585f6] hover:underline"
            >
              Consulter la version officielle sur Légifrance ↗
            </a>
          </div>
        </div>
      );
    },
  },
);
```

### B. Le Schéma Consolidation (`src/editor/schema.ts`)

```tsx
import { BlockNoteSchema, defaultBlockSpecs } from "@blocknote/core";
import { LawBlock } from "./blocks/LawBlock";

export const appSchema = BlockNoteSchema.create({
  blockSpecs: {
    ...defaultBlockSpecs,
    law: LawBlock,
  },
});

export type AppBlockNoteEditor = typeof appSchema.BlockNoteEditor;
```

### C. La Commande Slash (`src/editor/extensions/lawSlashMenu.ts`)

```tsx
import type { AppBlockNoteEditor } from "../schema";

export const createLawSlashItem = (editor: AppBlockNoteEditor) => ({
  title: "Texte de Loi (Légifrance)",
  onItemClick: async () => {
    // Insère le bloc de loi après le bloc actuel
    editor.insertBlocks(
      [
        {
          type: "law",
          props: {
            codeName: "Code général de la fonction publique",
            articleNumber: "Article L. 111-1",
            legifranceUrl:
              "https://www.legifrance.gouv.fr/codes/article_lc/LEGIARTI000044414167",
            excerpt:
              "Les agents publics sont au service de la Nation et de l'intérêt général.",
            status: "En vigueur",
          },
        },
      ],
      editor.getTextCursorPosition().block,
      "after",
    );
  },
  aliases: ["law", "loi", "legifrance", "article", "droit", "juridique"],
  group: "Outils de l'État",
  icon: <span>⚖️</span>,
  subtext: "Lier un texte de loi officiel certifié",
});
```

### D. L'Éditeur Complet (`src/editor/DocsEditor.tsx`)

```tsx
import "@blocknote/core/fonts/inter.css";
import "@blocknote/mantine/style.css";
import { BlockNoteView } from "@blocknote/mantine";
import {
  useCreateBlockNote,
  SuggestionMenuController,
  getDefaultSlashMenuItems,
} from "@blocknote/react";
import { appSchema } from "./schema";
import { createLawSlashItem } from "./extensions/lawSlashMenu";

export function DocsEditor({ ydoc, provider }: { ydoc: any; provider: any }) {
  const editor = useCreateBlockNote({
    schema: appSchema,
    collaboration: {
      provider,
      fragment: ydoc.getXmlFragment("document-store"),
      user: {
        name: "Agent DINUM",
        color: "#000091",
      },
    },
  });

  return (
    <BlockNoteView editor={editor} slashMenu={false}>
      <SuggestionMenuController
        triggerCharacter="/"
        getItems={async (query) => {
          const defaultItems = getDefaultSlashMenuItems(editor);
          const customItems = [createLawSlashItem(editor)];
          const allItems = [...defaultItems, ...customItems];

          return allItems.filter(
            (item) =>
              item.title.toLowerCase().includes(query.toLowerCase()) ||
              item.aliases?.some((a) =>
                a.toLowerCase().includes(query.toLowerCase()),
              ),
          );
        }}
      />
    </BlockNoteView>
  );
}
```
