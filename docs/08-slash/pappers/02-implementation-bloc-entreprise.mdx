## 💻 1. Déclaration du Composant React (`CompanyBlockSpec.tsx`)

```tsx
import { createReactBlockSpec } from "@blocknote/react";

export const CompanyBlockSpec = createReactBlockSpec(
  {
    type: "companyCard",
    propSchema: {
      siren: { default: "" },
      name: { default: "" },
      legalForm: { default: "" },
      address: { default: "" },
      nafCode: { default: "" },
      nafLabel: { default: "" },
      leaderName: { default: "" },
      revenue: { default: "" },
      inSolvency: { default: "false" },
      displayMode: { default: "card" }, // 'badge' | 'card' | 'financial' | 'full'
    },
    content: "none",
  },
  {
    render: (props) => {
      const { siren, name, legalForm, address, nafCode, nafLabel, leaderName, revenue, inSolvency, displayMode } = props.block.props;
      const isSolvent = inSolvency !== "true";

      // Mode Badge Compact
      if (displayMode === "badge") {
        return (
          <span className="fr-badge fr-badge--info fr-badge--sm">
            🏢 {name} (SIREN : {siren})
          </span>
        );
      }

      // Mode Fiche Synthétique DSFR
      return (
        <div className="fr-card fr-card--horizontal fr-card--grey my-4 p-4 border border-gray-200 dark:border-gray-800 rounded-xl bg-white dark:bg-gray-900 shadow-xs">
          <div className="flex flex-col gap-2 w-full">
            <div className="flex items-center justify-between">
              <div className="flex items-center gap-2">
                <span className="text-xl">🏢</span>
                <h4 className="text-base font-bold m-0">{name}</h4>
                <span className="fr-badge fr-badge--sm fr-badge--info">{legalForm}</span>
              </div>
              <span className={`fr-badge fr-badge--sm ${isSolvent ? "fr-badge--success" : "fr-badge--error"}`}>
                {isSolvent ? "In bonis (Régulier)" : "Procédure collective"}
              </span>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-3 text-xs text-gray-600 dark:text-gray-400 mt-2">
              <div>
                <strong>SIREN :</strong> {siren}
              </div>
              <div>
                <strong>Activité (NAF) :</strong> {nafCode} — {nafLabel}
              </div>
              <div>
                <strong>Dirigeant :</strong> {leaderName || "Non renseigné"}
              </div>
            </div>

            <div className="flex items-center justify-between text-xs text-gray-500 mt-2 pt-2 border-t border-gray-100 dark:border-gray-800">
              <span>📍 {address}</span>
              {revenue && <span>💶 CA : {revenue}</span>}
            </div>
          </div>
        </div>
      );
    },
  }
);
```

## 🐍 2. Proxy Django & Mise en Cache Redis (`companies/views.py`)

```python
from rest_framework.views import APIView
from rest_framework.response import Response
from django.core.cache import cache
import requests
from django.conf import settings

class CompanySearchView(APIView):
    def get(self, request):
        query = request.query_params.get("q", "").strip()
        if not query:
            return Response({"results": []})

        cache_key = f"company:search:{query}"
        cached = cache.get(cache_key)
        if cached:
            return Response({"results": cached})

        # Appel vers l'API Pappers ou API Entreprise
        url = "https://api.pappers.fr/v2/recherche"
        resp = requests.get(
            url,
            params={"q": query, "api_token": settings.PAPPERS_API_KEY, "par_page": 10},
            timeout=3.5,
        )
        data = resp.json()
        results = data.get("resultats", [])

        cache.set(cache_key, results, timeout=86400) # Cache 24h
        return Response({"results": results})
```
