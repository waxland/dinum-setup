L'éditeur de **La Suite Docs** s'appuie sur la bibliothèque **BlockNote.js** (fondée sur ProseMirror et TipTap). L'API `createReactBlockSpec` permet de déclarer un bloc personnalisé avec typage TypeScript complet.

---

## 💻 1. Déclaration du Schéma du Bloc (`LawBlock.tsx`)

```tsx
import { createReactBlockSpec } from "@blocknote/react";

export const LawBlockSpec = createReactBlockSpec(
  {
    type: "lawArticle",
    propSchema: {
      articleId: { default: "" },
      articleNum: { default: "" },
      codeTitle: { default: "" },
      contentHtml: { default: "" },
      status: { default: "VIGUEUR" },
      displayMode: { default: "citation" }, // 'badge' | 'citation' | 'card' | 'full'
      url: { default: "" },
      lastUpdate: { default: "" },
    },
    content: "none",
  },
  {
    render: (props) => {
      const { articleNum, codeTitle, contentHtml, status, displayMode, url } = props.block.props;

      // Mode Badge Inline
      if (displayMode === "badge") {
        return (
          <a
            href={url}
            target="_blank"
            rel="noopener noreferrer"
            className="fr-badge fr-badge--info fr-badge--sm no-underline"
          >
            ⚖️ {articleNum} — {codeTitle}
          </a>
        );
      }

      // Mode Citation / Callout DSFR
      return (
        <div className="fr-callout fr-callout--blue-france my-4">
          <h3 className="fr-callout__title">
            {articleNum} — {codeTitle}
          </h3>
          <p
            className="fr-callout__text text-sm italic"
            dangerouslySetInnerHTML={{ __html: contentHtml }}
          />
          <div className="mt-3 flex items-center justify-between text-xs text-gray-500">
            <span className="fr-badge fr-badge--success fr-badge--sm">{status}</span>
            <a href={url} target="_blank" rel="noopener noreferrer" className="fr-link">
              Consulter sur Légifrance ↗
            </a>
          </div>
        </div>
      );
    },
  }
);
```
