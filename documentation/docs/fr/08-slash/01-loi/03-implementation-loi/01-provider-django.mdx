import os
import requests
from django.conf import settings
from django.core.cache import cache
from ..base import BaseSourceProvider, SourceSearchResult

class LawSourceProvider(BaseSourceProvider):
    provider_id = "piste"
    source_type = "law"

    TOKEN_CACHE_KEY = "piste:auth:access_token"
    SUGGEST_URL = f"{getattr(settings, 'PISTE_API_BASE_URL', 'https://api.piste.gouv.fr/dila/legifrance/lf-engine-app')}/suggest"
    SEARCH_URL = f"{getattr(settings, 'PISTE_API_BASE_URL', 'https://api.piste.gouv.fr/dila/legifrance/lf-engine-app')}/search"

    def is_enabled(self) -> bool:
        client_id = getattr(settings, "PISTE_CLIENT_ID", "")
        client_secret = getattr(settings, "PISTE_CLIENT_SECRET", "")
        mock_enabled = getattr(settings, "PISTE_MOCK_ENABLED", False)
        return bool((client_id and client_secret) or mock_enabled)

    def _get_access_token(self) -> str:
        token = cache.get(self.TOKEN_CACHE_KEY)
        if token:
            return token

        resp = requests.post(
            settings.PISTE_OAUTH_TOKEN_URL,
            data={
                "grant_type": "client_credentials",
                "client_id": settings.PISTE_CLIENT_ID,
                "client_secret": settings.PISTE_CLIENT_SECRET,
                "scope": "openid",
            },
            timeout=4.0,
        )
        resp.raise_for_status()
        data = resp.json()
        token = data["access_token"]
        expires_in = data.get("expires_in", 3600)
        cache.set(self.TOKEN_CACHE_KEY, token, timeout=max(60, expires_in - 100))
        return token

    def search(self, query: str, limit: int = 5) -> list[SourceSearchResult]:
        if getattr(settings, "PISTE_MOCK_ENABLED", False):
            return self._get_mock_results(query)

        token = self._get_access_token()
        headers = {"Authorization": f"Bearer {token}", "Content-Type": "application/json"}
        payload = {
            "fond": "CODE_DATE",
            "recherche": {
                "champs": [{"typeChamp": "ALL", "criteres": [{"typeRecherche": "TOUS_LES_MOTS_DANS_UN_CHAMP", "valeur": query}]}],
                "pageNumber": 1,
                "pageSize": limit,
            }
        }
        resp = requests.post(self.SEARCH_URL, json=payload, headers=headers, timeout=3.5)
        resp.raise_for_status()
        articles = resp.json().get("results", [])

        results: list[SourceSearchResult] = []
        for a in articles:
            results.append({
                "source_id": a.get("id", ""),
                "source_type": "law",
                "provider": "Légifrance (PISTE)",
                "title": f"Article {a.get('num', '')}",
                "subtitle": a.get("codeTitle", "Code officiel"),
                "status": a.get("etat", "VIGUEUR"),
                "status_badge_color": "success" if a.get("etat") == "VIGUEUR" else "warning",
                "contentHtml": a.get("texteHtml", ""),
                "summary": a.get("resume", ""),
                "meta_field_1_label": "Code",
                "meta_field_1_value": a.get("codeTitle", ""),
                "meta_field_2_label": "Date d'effet",
                "meta_field_2_value": a.get("dateDebut", ""),
                "meta_field_3_label": "Identifiant",
                "meta_field_3_value": a.get("id", ""),
                "url": f"https://www.legifrance.gouv.fr/codes/article_lc/{a.get('id', '')}",
            })
        return results

    def _get_mock_results(self, query: str) -> list[SourceSearchResult]:
        return [
            {
                "source_id": "LEGIARTI000038814944",
                "source_type": "law",
                "provider": "Légifrance Mock",
                "title": "Article L. 111-1",
                "subtitle": "Code de la commande publique",
                "status": "VIGUEUR",
                "status_badge_color": "success",
                "contentHtml": "<p>Un acheteur ou une autorité concédante est un pouvoir adjudicateur...</p>",
                "summary": "Définition des pouvoirs adjudicateurs et entités adjudicatrices.",
                "meta_field_1_label": "Code",
                "meta_field_1_value": "Commande publique",
                "meta_field_2_label": "Date d'effet",
                "meta_field_2_value": "01/04/2019",
                "meta_field_3_label": "Identifiant",
                "meta_field_3_value": "LEGIARTI000038814944",
                "url": "https://www.legifrance.gouv.fr/codes/article_lc/LEGIARTI000038814944",
            }
        ]
